async function search(forceQuery = null) { const q = forceQuery || document.getElementById('search-input').value; if(!q) return; const btn = document.getElementById('search-btn'); btn.innerText = "Bypassing..."; let data = null; // We try the most robust gateways first const gateways = [ `https://pipedapi.kavin.rocks/search?q=${encodeURIComponent(q)}&filter=videos`, `https://invidious.flokinet.to/api/v1/search?q=${encodeURIComponent(q)}&type=video`, `https://yt.artemislena.eu/api/v1/search?q=${encodeURIComponent(q)}&type=video` ]; for (let url of gateways) { try { const res = await fetch(url); if (!res.ok) continue; const json = await res.json(); // Normalize data between Piped (content) and Invidious (json) data = json.content ? json.content.filter(i => i.type === 'stream') : json; if(data && data.length > 0) break; } catch(e) { console.log("Gateway busy, jumping to next..."); } } if(!data || data.length === 0) { // ULTIMATE EMERGENCY FALLBACK: If all scrapers fail, // we use a direct search link to a stable node. btn.innerText = "Retrying..."; try { const emergencyRes = await fetch(`https://iv.melmac.space/api/v1/search?q=${encodeURIComponent(q)}`); data = await emergencyRes.json(); } catch(e) { alert("External servers are heavy. Please wait 10 seconds and try one more time!"); btn.innerText = "Search"; return; } } const target = document.getElementById('results'); target.innerHTML = ''; playlistQueue = data.slice(0, 10).map(i => ({ id: i.videoId || i.id, title: i.title })); playlistQueue.forEach((item, idx) => { const d = document.createElement('div'); d.className = 'lib-item'; d.style.background = 'var(--glass-item)'; d.style.padding = '12px'; d.style.marginBottom = '8px'; d.innerHTML = `${item.title}`; d.onclick = () => playEngine(idx); target.appendChild(d); }); btn.innerText = "Search"; }
